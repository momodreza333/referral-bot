app = Client("ref_bot_final", bot_token=BOT_TOKEN)

# تابع تولید رفرال فیک
def generate_fake_ref(length=10):
    return ''.join(random.choices(string.ascii_letters + string.digits, k=length))

# دستور /start
@app.on_message(filters.command("start") & filters.private)
async def start(client, message):
    await message.reply_text(
        "🔥 **ربات رفرال فیک مینی‌آپ**\n\n"
        "لینک مینی‌آپ را با دستور زیر ارسال کنید:\n"
        "`/ref <لینک_مینی_آپ> <تعداد_رفرال>`\n\n"
        "مثال:\n"
        "`/ref https://t.me/miniapp?start=ref123 100`\n\n"
        "⚠️ **بدون نیاز به شماره تلفن یا ورود به تلگرام.**"
    )

# دستور /ref برای تولید رفرال
@app.on_message(filters.command("ref") & filters.private)
async def generate_ref(client, message):
    try:
        args = message.text.split()
        if len(args) < 3:
            await message.reply_text("❌ **فرمت اشتباه!**\n\nاز این فرمت استفاده کنید:\n`/ref <لینک_مینی_آپ> <تعداد_رفرال>`")
            return

        miniapp_link = args[1]
        ref_count = int(args[2])

        if ref_count > 1000:
            await message.reply_text("⚠️ حداکثر تعداد رفرال: 1000")
            return

        await message.reply_text(f"⏳ در حال تولید **{ref_count}** رفرال فیک برای لینک:\n`{miniapp_link}`")

        # تولید رفرال‌های فیک و باز کردن لینک
        for i in range(ref_count):
            fake_ref = generate_fake_ref()
            full_link = f"{miniapp_link}&ref={fake_ref}" if "?" in miniapp_link else f"{miniapp_link}?ref={fake_ref}"

            # ارسال رفرال به کانال/گروه (اختیاری)
            if 'CHANNEL_ID' in globals() and i % 10 == 0:
                try:
                    await client.send_message(CHANNEL_ID, f"🔗 رفرال جدید: {full_link}")
                except:
                    pass

            # باز کردن لینک بدون نیاز به مرورگر
            try:
                requests.get(full_link, timeout=5)
            except:
                pass

            if (i + 1) % 50 == 0:
                await message.reply_text(f"✅ **{i + 1}** رفرال تولید شد...")

        await message.reply_text(f"✅ **تکمیل شد!**\n\n**{ref_count}** رفرال فیک برای لینک زیر تولید شد:\n`{miniapp_link}`")

    except Exception as e:
        await message.reply_text(f"❌ خطا: {str(e)}")

# اجرای ربات
print("🤖 ربات در حال اجرا...")
app.run()
